CREATE TABLE usuarios (
    id BIGSERIAL PRIMARY KEY,
    email VARCHAR(150) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    apellido VARCHAR(150) NOT NULL,
    telefono VARCHAR(20),
    foto_perfil_url TEXT,
    foto_portada TEXT,
    biografia TEXT,
    etiqueta: VARCHAR(150) ---> esto es cuando quieran poner que son: artistas, comediantes, etc.
    redes_sociales JSONB DEFAULT '{}'::jsonb,
    rol rol_usuario_enum NOT NULL DEFAULT 'usuario',
    ubicacion ---> esta propiedad nos sirve para cuando los usuarios sean de tipo organizador y puedan ingresar la ubicacion de su establecimiento de forma opcional.
    (Si es un usuario organizador Tambien puede agregar miembros para gestionar lo eventos y darles permisos).
    estado estado_usuario_enum NOT NULL DEFAULT 'activo',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ NULL,
    deleted_by BIGINT NULL REFERENCES usuarios(id)
);


Ya no existen organizaciones, ahora solo son un rol mas de usuario llamado: organizador, para hacerlo mas simple y rapido el desarrollo del proyecto, posteriormente en una version mas adelante se podra implementar organizaciones, pero por el momento no.

CREATE TABLE miembros_organizacion (

CREATE TABLE eventos (
    id BIGSERIAL PRIMARY KEY,
    organizador_id BIGINT NOT NULL REFERENCES organizaciones(id) ON DELETE CASCADE,
    categoria_id INT NOT NULL REFERENCES categorias(id) ON DELETE RESTRICT,
    ubicacion_id BIGINT NOT NULL REFERENCES ubicaciones(id) ON DELETE RESTRICT,
    creado_por BIGINT NOT NULL REFERENCES usuarios(id) ON DELETE RESTRICT,
    titulo VARCHAR(150) NOT NULL,
    descripcion TEXT NOT NULL,
    fecha_inicio TIMESTAMPTZ NOT NULL,
    fecha_fin TIMESTAMPTZ NOT NULL,
    aforo INT NOT NULL DEFAULT 100,
    imagenes JSONB DEFAULT '[]'::jsonb,
    online  ---> este debe ser un bool.
    usuarios_cartelera  ---> es para agregar usuarios que son artistas y que pertenecen a la plataforma, si no existen de puede agregar un nombre y un link a una red social; maximo 5 artistas a la cartelera.
    restriccion_acceso VARCHAR(100) DEFAULT 'Todo público',
    etiquetas JSONB DEFAULT '[]'::jsonb,
    visibilidad NOT NULL DEFAULT 'publico',  --->como estado por defecto puede ser:oculto/privado
    localidades ---> aqui van tres campos: nombre de localidad, aforo y precio. esto debe ser calculado de acuerdo al aforo general del evento; maximo 4 localidades.
    informacion_pago ---> Esta información sirve para crear los mensajes automaticos de whatsapp para los eventos de pago, los campos son: nombre del destinatario, numero de contacto para generar el mensaje automatic, numero de cuenta bancaria, tipo de cuenta, numero de cedula; Aqui me gustaria agregar un paso de verificacion de datos como una foto del id o algo para poder hacerlo legal ya que pueden existir estafas y deseo evitar eso. 
    preguntas_frecuentes JSONB DEFAULT '[]'::jsonb,  --> los campos son titulo de pregunta y un campo de respuesta.
    revisado_por BIGINT NULL REFERENCES usuarios(id),
    motivo_rechazo TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ NULL
);
